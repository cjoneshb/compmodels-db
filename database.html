theory_opts = ["All theory groups", ...new Set(verified.map(d => d.theory_group))].sort()
arch_opts   = ["All architectures",  ...new Set(verified.map(d => d.architecture))].sort()
deriv_opts  = ["All", "Yes", "Partial", "No"]

viewof theory_filter = Inputs.select(theory_opts, { label: "Theory group" })
viewof arch_filter   = Inputs.select(arch_opts,   { label: "Architecture" })
viewof deriv_filter  = Inputs.select(deriv_opts,  { label: "Mechanism-derived?" })
viewof search_text   = Inputs.search(verified, {
  placeholder: "Search authors, behaviors, theories…",
  label: "Search"
})
filtered = {
  let result = search_text
  if (theory_filter !== "All theory groups")
    result = result.filter(d => d.theory_group === theory_filter)
  if (arch_filter !== "All architectures")
    result = result.filter(d => d.architecture === arch_filter)
  if (deriv_filter !== "All")
    result = result.filter(d => d.mechanism_derived === deriv_filter)
  return result
}
function badge(val) {
  if (val === "Yes")     return "✓ Yes"
  if (val === "Partial") return "◑ Partial"
  if (val === "No")      return "✗ No"
  return val || "—"
}

arch_counts = {
  const counts = d3.rollups(
    verified, v => v.length, d => d.architecture
  ).map(([architecture, count]) => ({ architecture, count }))
  return counts.sort((a, b) => b.count - a.count)
}

theory_counts = {
  const counts = d3.rollups(
    verified, v => v.length, d => d.theory_group
  ).map(([theory_group, count]) => ({ theory_group, count }))
  return counts.sort((a, b) => b.count - a.count)
}

    count: verified.filter(d => d.mechanism_derived === v).length,
    fill: colors[v]
  }))
}
